@dataclass(frozen=True)
class PmtpTensorDescriptor:
    """
    Fixed-width 88-byte C ABI Descriptor for High-Dimensional Tensors.
    Eliminates 1D token serialization for D >= 100.
    """
    segment_id: int
    buffer_id: int
    generation: int = 0
    byte_offset: int = 0
    byte_length: int = 0
    shape: Tuple[int, ...] = ()
    dtype_code: int = _DTYPE_F64
    magic: bytes = PMTP_MAGIC
    version: int = VERSION

    def __init__(
        self,
        segment_id: int = 0,
        buffer_id: int = 0,
        generation: int = 0,
        byte_offset: int = 0,
        byte_length: int = 0,
        shape: Tuple[int, ...] = (),
        dtype_code: int = _DTYPE_F64,
        offset: Optional[int] = None,
        magic: bytes = PMTP_MAGIC,
        version: int = VERSION,
    ) -> None:
        if offset is not None:
            byte_offset = offset
        sh = tuple(int(x) for x in shape)
        for dim in sh:
            if dim <= 0:
                raise ValueError(f"Shape dimensions must be strictly positive, got {dim} in {sh}")

        if byte_length == 0 and sh:
            itemsize = 4 if dtype_code == 2 else 8
            elems = math.prod(sh)
            byte_length = elems * itemsize

        object.__setattr__(self, "segment_id", int(segment_id))
        object.__setattr__(self, "buffer_id", int(buffer_id))
        object.__setattr__(self, "generation", int(generation))
        object.__setattr__(self, "byte_offset", int(byte_offset))
        object.__setattr__(self, "byte_length", int(byte_length))
        object.__setattr__(self, "shape", sh)
        object.__setattr__(self, "dtype_code", int(dtype_code))
        object.__setattr__(self, "magic", magic)
        object.__setattr__(self, "version", int(version))

        if not (1 <= len(self.shape) <= _MAX_DESCRIPTOR_NDIM):
            raise ValueError(f"shape rank must be 1..{_MAX_DESCRIPTOR_NDIM}")
        itemsize = 4 if self.dtype_code == 2 else 8
        expected = math.prod(self.shape) * itemsize
        if expected != self.byte_length:
            raise ValueError(f"byte_length mismatch: shape implies {expected} bytes, got {self.byte_length}")

    @property
    def ndim(self) -> int:
        return len(self.shape)

    @property
    def offset(self) -> int:
        return self.byte_offset

    def pack(self) -> bytes:
        padded_shape = tuple(int(x) for x in self.shape) + (0,) * (_MAX_DESCRIPTOR_NDIM - self.ndim)
        reserved_bytes = b"\x00" * 6
        return struct.pack(
            PMTP_DESCRIPTOR_FORMAT,
            PMTP_MAGIC,
            self.version,
            PMTP_DESCRIPTOR_SIZE,
            int(self.segment_id),
            int(self.buffer_id),
            int(self.generation),
            int(self.byte_offset),
            int(self.byte_length),
            int(self.dtype_code),
            int(self.ndim),
            reserved_bytes,
            *padded_shape,
        )

    @classmethod
    def unpack(cls, buf: bytes) -> PmtpTensorDescriptor:
        if len(buf) < PMTP_DESCRIPTOR_SIZE:
            raise ValueError(f"Descriptor buffer too short: {len(buf)} < {PMTP_DESCRIPTOR_SIZE}")
        parts = struct.unpack(PMTP_DESCRIPTOR_FORMAT, buf[:PMTP_DESCRIPTOR_SIZE])
        magic = parts[0]
        ver = parts[1]
        encoded_size = parts[2]
        if magic != PMTP_MAGIC:
            raise ValueError(f"Invalid PMTP magic: {magic} != {PMTP_MAGIC}")
        if encoded_size != PMTP_DESCRIPTOR_SIZE:
            raise ValueError(f"Encoded size mismatch: {encoded_size} != {PMTP_DESCRIPTOR_SIZE}")
        if ver not in (VERSION, 26154, 26153, 26152, 26151, 26150):
            raise ValueError(f"PMTP version mismatch: {ver} not compatible with {VERSION}")

        seg_id = parts[3]
        buf_id = parts[4]
        gen = parts[5]
        offset = parts[6]
        length = parts[7]
        dtype = parts[8]
        ndim = parts[9]
        reserved = parts[10]
        if any(b != 0 for b in reserved):
            raise ValueError("PMTP descriptor reserved bytes must be zero")

        if not (1 <= ndim <= _MAX_DESCRIPTOR_NDIM):
            raise ValueError(f"Invalid rank in descriptor: {ndim}")
        shape = tuple(parts[11 : 11 + ndim])

        # Validate that unassigned shape slots are zero
        for s in parts[11 + ndim : 15]:
            if s != 0:
                raise ValueError("Unused shape slots must be zero")

        return cls(
            segment_id=seg_id,
            buffer_id=buf_id,
            generation=gen,
            byte_offset=offset,
            byte_length=length,
            shape=shape,
            dtype_code=dtype,
            version=ver,
        )


class AnchoredArray(np.ndarray):
    """
    Theorem 691: Subclass of np.ndarray that anchors the underlying PmtpSlabAllocator
    in `_slab_owner`. Guarantees that even if user code drops the slab reference,
    Python GC cannot collect the allocator and trigger `mmap.close()` while this
    array view is alive. Eliminates SIGSEGV 139.
    """
    _slab_owner: Any = None


class PmtpSlabAllocator:
    """
    Cross-process, zero-copy shared memory slab manager.
    Hardened against path traversal, negative offsets, and concurrent truncation.
    """
    def __init__(self, slab_name: str, size_bytes: int, create: bool = True):
        self._mmap_obj: Optional[mmap.mmap] = None
        self._fd: Optional[int] = None

        if not isinstance(slab_name, str) or not SLAB_NAME_REGEX.match(slab_name):
            raise ValueError(f"Invalid slab_name (security regex failed): {slab_name!r}")
        if size_bytes <= 0:
            raise ValueError(f"size_bytes must be strictly positive, got {size_bytes}")

        self.slab_name = slab_name
        self.size_bytes = int(size_bytes)
        self.create = create

        if os.name == "nt":
            # Windows Named Shared Memory (tagname)
            self._mmap_obj = mmap.mmap(-1, self.size_bytes, tagname=self.slab_name)
        else:
            # POSIX /dev/shm shared memory with 0600 permissions
            shm_path = f"/dev/shm/{self.slab_name}"
            if create:
                fd = os.open(shm_path, os.O_CREAT | os.O_EXCL | os.O_RDWR, 0o600)
                try:
                    os.ftruncate(fd, self.size_bytes)
                except Exception:
                    os.close(fd)
                    if os.path.exists(shm_path):
                        os.unlink(shm_path)
                    raise
            else:
                fd = os.open(shm_path, os.O_RDWR)
            self._fd = fd
            self._mmap_obj = mmap.mmap(fd, self.size_bytes)

    def get_buffer(self) -> memoryview:
        if self._mmap_obj is None:
            raise RuntimeError("PMTP slab not mapped")
        return memoryview(self._mmap_obj)

    def as_ndarray(self, shape: Tuple[int, ...], dtype=np.float64, offset: int = 0) -> np.ndarray:
        if offset < 0:
            raise ValueError(f"offset cannot be negative: {offset}")
        for dim in shape:
            if dim <= 0:
                raise ValueError(f"Shape dimensions must be strictly positive: {shape}")
        dt = np.dtype(dtype)
        if dt.kind == 'O' or dt.hasobject:
            raise ValueError("Object dtypes containing python pointers are strictly disallowed in shared memory PMTP")
        if offset % dt.alignment != 0:
            raise ValueError(f"Offset {offset} is not aligned to dtype boundary ({dt.alignment} bytes)")
        itemsize = dt.itemsize

        # Theorem 691: Use math.prod for arbitrary-precision integer calculation
        count = math.prod(shape)
        nbytes = count * itemsize
        if offset + nbytes > self.size_bytes:
            raise ValueError(f"Array exceeds slab boundary: {offset + nbytes} > {self.size_bytes}")

        # Construct raw array from buffer and wrap in AnchoredArray
        raw_arr = np.ndarray(shape=shape, dtype=dtype, buffer=self.get_buffer(), offset=offset)
        anchored = raw_arr.view(AnchoredArray)
        anchored._slab_owner = self
        return anchored

    def close(self) -> None:
        # Theorem 695: RAII guard for partially-constructed objects.
        # If __init__ raises after object creation but before _mmap_obj assignment,
        # __del__ will trigger close() → AttributeError. Use hasattr() as O(1) guard.
        if not hasattr(self, "_mmap_obj"):
            return
        if self._mmap_obj is not None:
            try:
                self._mmap_obj.close()
            except BufferError:
                pass
            self._mmap_obj = None
        if not hasattr(self, "_fd"):
            return
        if self._fd is not None:
            try:
                os.close(self._fd)
            except Exception:
                pass
            self._fd = None

    def unlink(self) -> None:
        self.close()
        if os.name != "nt":
            shm_path = f"/dev/shm/{self.slab_name}"
            if os.path.exists(shm_path):
                try:
                    os.unlink(shm_path)
                except OSError:
                    pass

    def __enter__(self) -> "PmtpSlabAllocator":
        return self

    def __exit__(self, exc_type, exc_val, exc_tb) -> None:
        self.close()

    def __del__(self) -> None:
        self.close()

@dataclass

    vendor: str

class HardwareProbe:
    """
    Dynamically interrogates runtime environment without hardcoded vendor-lock.
    Uses cached inspection to eliminate heavy import overhead.
    """
    _cached_profile: Optional[HardwareProfile] = None

    @classmethod
    def probe(cls, force_refresh: bool = False) -> HardwareProfile:
        if cls._cached_profile is not None and not force_refresh:
            return cls._cached_profile

        available: List[str] = ["reference", "cpp", "rust"]
        vendor = "generic"
        target = "cpu_numa"
        mem_class = 3
        primary = "reference"
        dev_name = "CPU Generic Host"
        dev_count = os.cpu_count() or 1

        if "torch" in sys.modules:
            torch = sys.modules["torch"]
            if torch.cuda.is_available():
                vendor = "nvidia"
                target = "nvidia_cuda"
                mem_class = 2
                primary = "cuda"
                available.append("cuda")
                dev_name = torch.cuda.get_device_name(0)
                dev_count = torch.cuda.device_count()

        script_dir = Path(__file__).resolve().parent
        for cand in [script_dir, script_dir.parent / "respuestas", script_dir.parent]:
            if list(cand.glob("*rust*.dll")) or list(cand.glob("*rust*.so")):
                if "rust_native" not in available:
                    available.append("rust_native")
            if list(cand.glob("*cpp*.dll")) or list(cand.glob("*cpp*.so")):
                if "cpp_native" not in available:
                    available.append("cpp_native")

        if primary == "reference" and "rust_native" in available:
            primary = "rust_native"

        cls._cached_profile = HardwareProfile(
            vendor=vendor,
            target=target,
            memory_class=mem_class,
            primary_backend=primary,
            available_backends=available,
            device_count=dev_count,
            device_name=dev_name,
        )
        return cls._cached_profile

def cayley_direct_oracle_dxd(X: np.ndarray, Z: np.ndarray, step: float) -> np.ndarray:
    """
    Exact Differential Cayley Oracle in Full D x D Dimension:
    A = step * (Z X^T - X Z^T)
    Y = (I - 0.5 A)^{-1} (I + 0.5 A) X
    Used as an unyielding mathematical oracle to guarantee that identity stubs (Y=X)
    or geometrically deviated solutions can NEVER pass certification.
    """
    D, K = X.shape
    if D > 2000:
        raise ValueError(f"cayley_direct_oracle_dxd is restricted to D <= 2000 to prevent OOM crash (got D={D})")
    A = step * (Z @ X.T - X @ Z.T)
    I_D = np.eye(D, dtype=np.float64)
    return np.linalg.solve(I_D - 0.5 * A, (I_D + 0.5 * A) @ X)

        # Lyapunov 1: G C + C G = S + S.T
        rhs_sym = S + S.T
        Q_rhs_Q = Q.T @ rhs_sym @ Q
        C_eig = Q_rhs_Q / lam_sum
        C = Q @ C_eig @ Q.T
        C = 0.5 * (C + C.T)

        St = S - G @ C
        St = 0.5 * (St - St.T)

        # Lyapunov 2: G B + B G = -St
        Q_st_Q = Q.T @ (-St) @ Q
        B_eig = Q_st_Q / lam_sum
        B = Q @ B_eig @ Q.T
        B = 0.5 * (B - B.T)

        Ez = Ginv
        Ex = (B - C) @ Ginv

        XtP = G @ Ex + S @ Ez
        PtP = (
            Ex.T @ G @ Ex +
            Ex.T @ S @ Ez +
            Ez.T @ S.T @ Ex +
            Ez.T @ T @ Ez
        )
        PtP = 0.5 * (PtP + PtP.T)

        scale = max(1.0, abs(effective_step))
        inv_scale = 1.0 / scale
        tau = effective_step / scale
        half_tau = 0.5 * tau

        M_top_left = inv_scale * np.eye(K, dtype=np.float64) - half_tau * XtP
        M_top_right = half_tau * G
        M_bot_left = -half_tau * PtP
        M_bot_right = inv_scale * np.eye(K, dtype=np.float64) + half_tau * XtP.T

        M = np.block([
            [M_top_left,  M_top_right],
            [M_bot_left,  M_bot_right]
        ])

        RHS = np.block([
            [tau * G],
            [tau * XtP.T]
        ])

        try:
            W = np.linalg.solve(M, RHS)
        except np.linalg.LinAlgError as e:
            return Status.NUMERICAL_INSTABILITY, None, {"error": f"solve failed (singular M): {e}"}

        if not np.all(np.isfinite(W)):
            return Status.NUMERICAL_INSTABILITY, None, {"error": "solve returned non-finite W"}

        W1 = W[:K, :]
        W2 = W[K:, :]

        Ax = np.eye(K, dtype=np.float64) + Ex @ W1 - W2
        Az = Ez @ W1

        if out is None:
            out = np.empty_like(X)

        # Memory allocation optimization: direct in-place accumulation
        np.dot(X, Ax, out=out)
        out += Z_unit @ Az

        # 6. Invariant audits & Recalculation post-fallback
        G_out = out.T @ out
        ortho_defect = float(np.max(np.abs(G_out - np.eye(K))))
        gram_drift = float(np.max(np.abs(G_out - G)))
        fallback_engaged = False

        if math.isnan(ortho_defect) or math.isinf(ortho_defect):
            return Status.NUMERICAL_FAILURE, None, {"error": "Non-finite orthogonality defect"}

        if force_fallback or not (ortho_defect <= 1.0e-11):
            out[:] = self.polar_qr_reanchor(out)
            G_out = out.T @ out
            ortho_defect = float(np.max(np.abs(G_out - np.eye(K))))
            gram_drift = float(np.max(np.abs(G_out - G)))
            fallback_engaged = True
            if ortho_defect > OUTPUT_TOLERANCE:
                return Status.NUMERICAL_FAILURE, None, {"error": f"Post-fallback orthogonality defect unacceptable: {ortho_defect:.3e}"}

        metrics = {
            "ortho_defect": ortho_defect,
            "gram_drift": gram_drift,
            "step": step,
            "D": D,
            "K": K,
            "fallback_engaged": fallback_engaged,
            "backend": self.active_backend
        }

        return Status.OK, out, metrics

// Neumaier compensated summation with strict FP ordering (Theorem 690: no volatile, enabling SIMD vectorization)
inline void neumaier_accumulate(double val, double& sum, double& comp) noexcept {
    double s = sum;
    double v = val;
    double t = s + v;
    if (std::abs(s) >= std::abs(v)) {
        double diff = s - t;
        comp += diff + v;
    } else {
        double diff = v - t;
        comp += diff + s;
    }
    sum = t;
}

#pragma GCC pop_options
#endif

inline double dot_neumaier(const double* a, const double* b, std::uint64_t n) noexcept {
    double sum = 0.0;
    double comp = 0.0;
    for (std::uint64_t i = 0; i < n; ++i) {
        double prod = a[i] * b[i];
        neumaier_accumulate(prod, sum, comp);
    }
    return sum + comp;
}

// Universal buffer overlap checker: overlap = (start_a < end_b) && (start_b < end_a)
inline bool buffers_overlap(const double* a, std::uint64_t len_a, const double* b, std::uint64_t len_b) noexcept {
    if (!a || !b || len_a == 0 || len_b == 0) return false;
    auto start_a = reinterpret_cast<std::uintptr_t>(a);
    auto start_b = reinterpret_cast<std::uintptr_t>(b);
    std::uintptr_t bytes_a = len_a * sizeof(double);
    std::uintptr_t bytes_b = len_b * sizeof(double);
    if (UINTPTR_MAX - start_a < bytes_a || UINTPTR_MAX - start_b < bytes_b) {
        return true; // Pointer arithmetic overflow defense
    }
    auto end_a = start_a + bytes_a;
    auto end_b = start_b + bytes_b;
    return (start_a < end_b) && (start_b < end_a);
}

inline bool is_unsafe_partial_alias(const double* a, std::uint64_t len_a, const double* b, std::uint64_t len_b) noexcept {
    return buffers_overlap(a, len_a, b, len_b);
}

// Spherical Projection: y = x / ||x||_2 with Two-Pass Scaled Norm and OpenMP Acceleration
POLYDIM_EXPORT std::int32_t polydim_sphere_project_v26155(
    PolydimF64ConstView in_view,
    PolydimF64MutView out_view
) noexcept {
    if (!in_view.ptr || !out_view.ptr) return POLYDIM_INVALID_ARGUMENT;
    if (in_view.len != out_view.len)   return POLYDIM_LENGTH_MISMATCH;
    if (in_view.len == 0 || in_view.len > kMaxDimension) return POLYDIM_RANGE_ERROR;

    // Strict 8-byte pointer alignment verification
    if (reinterpret_cast<std::uintptr_t>(in_view.ptr) % alignof(double) != 0 ||
        reinterpret_cast<std::uintptr_t>(out_view.ptr) % alignof(double) != 0) {
        return POLYDIM_ALIGNMENT_ERROR;
    }

    bool is_in_place = (in_view.ptr == out_view.ptr);
    if (!is_in_place && buffers_overlap(in_view.ptr, in_view.len, out_view.ptr, out_view.len)) {
        return POLYDIM_ALIAS_VIOLATION;
    }

    std::uint64_t n = in_view.len;

    // Pass 1: Find max absolute value to avoid overflow (1e308) and underflow (1e-308)
    double max_val = 0.0;
    for (std::uint64_t i = 0; i < n; ++i) {
        double v = std::abs(in_view.ptr[i]);
        if (!std::isfinite(v)) return POLYDIM_NONFINITE_INPUT;
        if (v > max_val) max_val = v;
    }

    if (max_val == 0.0) return POLYDIM_ZERO_VECTOR;

    // Pass 2: Binary exponent scaling with DAZ/FTZ safe factor
    double scale = 1.0;
    double inv_scale = 1.0;
    if (max_val < std::numeric_limits<double>::min()) {
        scale = 1.0e300;
        inv_scale = 1.0 / (max_val * scale);
    } else if (max_val > 1.0e300) {
        scale = 1.0e-150;
        inv_scale = 1.0 / (max_val * scale);
    } else {
        scale = 1.0;
        inv_scale = 1.0 / max_val;
    }

    double sum = 0.0;
    double comp = 0.0;
    for (std::uint64_t i = 0; i < n; ++i) {
        double scaled = (in_view.ptr[i] * scale) * inv_scale;
        neumaier_accumulate(scaled * scaled, sum, comp);
    }
    double scaled_norm_sq = sum + comp;
    if (!std::isfinite(scaled_norm_sq) || scaled_norm_sq <= 0.0) return POLYDIM_NUMERICAL_FAILURE;

    double scaled_norm = std::sqrt(scaled_norm_sq);
    double inv_scaled_norm = 1.0 / scaled_norm;
    if (!std::isfinite(inv_scaled_norm) || inv_scaled_norm <= 0.0) return POLYDIM_NUMERICAL_FAILURE;

    // Write phase with inline validation and OpenMP (Theorem 700 associative scaling)
    #if defined(_OPENMP)
    #pragma omp parallel for if(n >= 10000) schedule(static)
    #endif
    for (std::int64_t i = 0; i < static_cast<std::int64_t>(n); ++i) {
        out_view.ptr[i] = ((in_view.ptr[i] * scale) * inv_scale) * inv_scaled_norm;
    }

    return POLYDIM_OK;
}

// Spherical Parallel Transport on S^(D-1)
POLYDIM_EXPORT std::int32_t polydim_sphere_transport_v26155(
    PolydimF64ConstView x_view,
    PolydimF64ConstView y_view,
    PolydimF64ConstView v_view,
    PolydimF64MutView out_view
) noexcept {
    if (!x_view.ptr || !y_view.ptr || !v_view.ptr || !out_view.ptr) return POLYDIM_INVALID_ARGUMENT;
    std::uint64_t n = x_view.len;
    if (y_view.len != n || v_view.len != n || out_view.len != n) return POLYDIM_LENGTH_MISMATCH;
    if (n == 0 || n > kMaxDimension) return POLYDIM_RANGE_ERROR;

    // Strict 8-byte alignment verification
    if (reinterpret_cast<std::uintptr_t>(x_view.ptr) % alignof(double) != 0 ||
        reinterpret_cast<std::uintptr_t>(y_view.ptr) % alignof(double) != 0 ||
        reinterpret_cast<std::uintptr_t>(v_view.ptr) % alignof(double) != 0 ||
        reinterpret_cast<std::uintptr_t>(out_view.ptr) % alignof(double) != 0) {
        return POLYDIM_ALIGNMENT_ERROR;
    }

    // Complete pairwise anti-aliasing validation (rejects any overlap with out_view and pairwise inputs)
    if (buffers_overlap(x_view.ptr, n, out_view.ptr, n) ||
        buffers_overlap(y_view.ptr, n, out_view.ptr, n) ||
        buffers_overlap(v_view.ptr, n, out_view.ptr, n) ||
        buffers_overlap(x_view.ptr, n, y_view.ptr, n) ||
        buffers_overlap(x_view.ptr, n, v_view.ptr, n) ||
        buffers_overlap(y_view.ptr, n, v_view.ptr, n)) {
        return POLYDIM_ALIAS_VIOLATION;
    }

    double tol_d = get_dimension_tolerance(n);

    // Check unit norm of base points with NaN-safe guard
    double x_norm_sq = dot_neumaier(x_view.ptr, x_view.ptr, n);
    double y_norm_sq = dot_neumaier(y_view.ptr, y_view.ptr, n);
    if (!std::isfinite(x_norm_sq) || !std::isfinite(y_norm_sq)) {
        return POLYDIM_NONFINITE_INPUT;
    }
    if (std::abs(x_norm_sq - 1.0) > tol_d || std::abs(y_norm_sq - 1.0) > tol_d) {
        return POLYDIM_NOT_UNIT;
    }

    // Relative tangency check
    double v_norm_sq = dot_neumaier(v_view.ptr, v_view.ptr, n);
    if (!std::isfinite(v_norm_sq)) return POLYDIM_NONFINITE_INPUT;
    double v_norm = std::sqrt(std::max(0.0, v_norm_sq));

    double xt_v = dot_neumaier(x_view.ptr, v_view.ptr, n);
    if (!std::isfinite(xt_v)) {
        return POLYDIM_NONFINITE_INPUT;
    }
    double tol_tangency = tol_d * std::max(1.0, v_norm);
    if (std::abs(xt_v) > tol_tangency) {
        return POLYDIM_NOT_TANGENT;
    }

    // Dot product: cos(theta) = x^T y
    double cos_th = dot_neumaier(x_view.ptr, y_view.ptr, n);
    if (!std::isfinite(cos_th)) {
        return POLYDIM_NONFINITE_INPUT;
    }
    cos_th = std::clamp(cos_th, -1.0, 1.0);

    // Antipodal check
    if (1.0 + cos_th <= kCutLocusNormSq) {
        return POLYDIM_PATH_REQUIRED;
    }

    // Standard Levi-Civita Transport factor
    double yt_v = dot_neumaier(y_view.ptr, v_view.ptr, n);
    if (!std::isfinite(yt_v)) {
        return POLYDIM_NONFINITE_INPUT;
    }
    double factor = yt_v / (1.0 + cos_th);

    #if defined(_OPENMP)
    #pragma omp parallel for if(n >= 10000) schedule(static)
    #endif
    for (std::int64_t i = 0; i < static_cast<std::int64_t>(n); ++i) {
        out_view.ptr[i] = v_view.ptr[i] - factor * (x_view.ptr[i] + y_view.ptr[i]);
    }

    // Output relative tangency audit
    double yt_out = dot_neumaier(y_view.ptr, out_view.ptr, n);
    if (!std::isfinite(yt_out) || std::abs(yt_out) > (tol_d * 100.0) * std::max(1.0, v_norm)) {
        return POLYDIM_PRECISION_LOSS;
    }

    // Isometry audit: ||out|| == ||v||
    double out_norm_sq = dot_neumaier(out_view.ptr, out_view.ptr, n);
    if (!std::isfinite(out_norm_sq)) return POLYDIM_PRECISION_LOSS;
    double out_norm = std::sqrt(std::max(0.0, out_norm_sq));
    if (std::abs(out_norm - v_norm) > (tol_d * 100.0) * std::max(1.0, v_norm)) {
        return POLYDIM_PRECISION_LOSS;
    }

    return POLYDIM_OK;
}

        if (expected_elements / desc->shape[i] != prev) {
            return POLYDIM_RANGE_ERROR;
        }
    }

    for (std::uint8_t i = desc->ndim; i < 4; ++i) {
        if (desc->shape[i] != 0) return POLYDIM_RANGE_ERROR;

        desc->byte_length % sizeof(double) != 0) {
        return POLYDIM_LENGTH_MISMATCH;
    }

        return POLYDIM_ALIGNMENT_ERROR;
    }

// Neumaier Compensated Summation
#[inline(always)]
pub fn neumaier_sum(slice: &[f64]) -> f64 {
    let mut sum = 0.0f64;
    let mut comp = 0.0f64;
    for &val in slice {
        let t = sum + val;
        if sum.abs() >= val.abs() {
            comp += (sum - t) + val;
        } else {
            comp += (val - t) + sum;
        }
        sum = t;
    }
    sum + comp
}

#[inline(always)]
pub fn dot_neumaier(a: &[f64], b: &[f64]) -> f64 {
    let mut sum = 0.0f64;
    let mut comp = 0.0f64;
    let len = a.len().min(b.len());
    for i in 0..len {
        let prod = a[i] * b[i];
        let t = sum + prod;
        if sum.abs() >= prod.abs() {
            comp += (sum - t) + prod;
        } else {
            comp += (prod - t) + sum;
        }
        sum = t;
    }
    sum + comp
}

#[inline(always)]
pub fn buffers_overlap(ptr_a: *const f64, len_a: usize, ptr_b: *const f64, len_b: usize) -> bool {
    if ptr_a.is_null() || ptr_b.is_null() || len_a == 0 || len_b == 0 {
        return false;
    }
    let a0 = ptr_a as usize;
    let b0 = ptr_b as usize;
    let a_bytes = len_a.saturating_mul(size_of::<f64>());
    let b_bytes = len_b.saturating_mul(size_of::<f64>());
    let a1 = a0.saturating_add(a_bytes);
    let b1 = b0.saturating_add(b_bytes);

    a0 < b1 && b0 < a1
}

#[no_mangle]

    in_view: PolydimF64ConstView,
    out_view: PolydimF64MutView,

    let result = catch_unwind(AssertUnwindSafe(|| {
        if in_view.ptr.is_null() || out_view.ptr.is_null() {
            return POLYDIM_INVALID_ARGUMENT;

            return POLYDIM_ALIGNMENT_ERROR;

            return POLYDIM_ALIAS_VIOLATION;

                    return POLYDIM_NONFINITE_INPUT;

                    max_val = abs_x;
                }

                return POLYDIM_ZERO_VECTOR;

            let mut comp = 0.0f64;
            for &x in slice.iter() {
                let scaled = (x * scale) * inv_scale;
                let prod = scaled * scaled;
                let t = sum + prod;
                if sum.abs() >= prod.abs() {
                    comp += (sum - t) + prod;
                } else {
                    comp += (prod - t) + sum;
                }
                sum = t;
            }
            let scaled_norm_sq = sum + comp;
            if !scaled_norm_sq.is_finite() || scaled_norm_sq <= 0.0 {
                return POLYDIM_NUMERICAL_FAILURE;
            }

            let scaled_norm = scaled_norm_sq.sqrt();
            let inv_scaled_norm = 1.0 / scaled_norm;
            if !inv_scaled_norm.is_finite() || inv_scaled_norm <= 0.0 {
                return POLYDIM_NUMERICAL_FAILURE;
            }

            for val in slice.iter_mut() {
                let res = ((*val * scale) * inv_scale) * inv_scaled_norm;
                if !res.is_finite() {
                    return POLYDIM_NUMERICAL_FAILURE;
                }
                *val = res;
            }
        } else {
            let in_slice = unsafe { std::slice::from_raw_parts(in_view.ptr, n) };
            let out_slice = unsafe { std::slice::from_raw_parts_mut(out_view.ptr, n) };

            // Pass 1: Max absolute value
            let mut max_val = 0.0f64;
            for &x in in_slice.iter() {
                if !x.is_finite() {
                    return POLYDIM_NONFINITE_INPUT;
                }
                let abs_x = x.abs();
                if abs_x > max_val {
                    max_val = abs_x;
                }
            }

            if max_val == 0.0 {
                return POLYDIM_ZERO_VECTOR;
            }

            // Pass 2: Binary exponent scaling for FP64 subnormals & extreme inputs
            let (scale, inv_scale) = if max_val < f64::MIN_POSITIVE || is_subnormal_raw_bits(max_val) {
                let s = 1.0e300f64;
                (s, 1.0 / (max_val * s))
            } else if max_val > 1.0e300 {
                let s = 1.0e-150f64;
                (s, 1.0 / (max_val * s))
            } else {
                (1.0, 1.0 / max_val)
            };

            let mut sum = 0.0f64;
            let mut comp = 0.0f64;
            for &x in in_slice.iter() {
                let scaled = (x * scale) * inv_scale;
                let prod = scaled * scaled;
                let t = sum + prod;
                if sum.abs() >= prod.abs() {
                    comp += (sum - t) + prod;
                } else {
                    comp += (prod - t) + sum;
                }
                sum = t;
            }
            let scaled_norm_sq = sum + comp;
            if !scaled_norm_sq.is_finite() || scaled_norm_sq <= 0.0 {
                return POLYDIM_NUMERICAL_FAILURE;
            }

            let scaled_norm = scaled_norm_sq.sqrt();
            let inv_scaled_norm = 1.0 / scaled_norm;
            if !inv_scaled_norm.is_finite() || inv_scaled_norm <= 0.0 {
                return POLYDIM_NUMERICAL_FAILURE;
            }

            for (dst, &src) in out_slice.iter_mut().zip(in_slice.iter()) {
                let res = ((src * scale) * inv_scale) * inv_scaled_norm;
                if !res.is_finite() {
                    return POLYDIM_NUMERICAL_FAILURE;
                }
                *dst = res;
            }
        }

        POLYDIM_OK
    }));

    result.unwrap_or(POLYDIM_PANIC_TRAPPED)
}

// Spherical Parallel Transport on S^(D-1)
#[no_mangle]
pub unsafe extern "C" fn polydim_sphere_transport_v26155(
    x_view: PolydimF64ConstView,
    y_view: PolydimF64ConstView,
    v_view: PolydimF64ConstView,
    out_view: PolydimF64MutView,
) -> i32 {
    let result = catch_unwind(AssertUnwindSafe(|| {
        if x_view.ptr.is_null() || y_view.ptr.is_null() || v_view.ptr.is_null() || out_view.ptr.is_null() {
            return POLYDIM_INVALID_ARGUMENT;
        }
        let n_u64 = x_view.len;
        if y_view.len != n_u64 || v_view.len != n_u64 || out_view.len != n_u64 {
            return POLYDIM_LENGTH_MISMATCH;
        }
        if n_u64 == 0 || n_u64 > MAX_DIMENSION || n_u64 > (isize::MAX as u64) {
            return POLYDIM_RANGE_ERROR;
        }
        let n = n_u64 as usize;

        // Alignment check
        if (x_view.ptr as usize) % align_of::<f64>() != 0 ||
           (y_view.ptr as usize) % align_of::<f64>() != 0 ||
           (v_view.ptr as usize) % align_of::<f64>() != 0 ||
           (out_view.ptr as usize) % align_of::<f64>() != 0 {
            return POLYDIM_ALIGNMENT_ERROR;
        }

        // Complete pairwise anti-aliasing check (rejects any overlap with out_view and pairwise inputs)
        let out_const = out_view.ptr as *const f64;
        if buffers_overlap(x_view.ptr, n, out_const, n) ||
           buffers_overlap(y_view.ptr, n, out_const, n) ||
           buffers_overlap(v_view.ptr, n, out_const, n) ||
           buffers_overlap(x_view.ptr, n, y_view.ptr, n) ||
           buffers_overlap(x_view.ptr, n, v_view.ptr, n) ||
           buffers_overlap(y_view.ptr, n, v_view.ptr, n) {
            return POLYDIM_ALIAS_VIOLATION;
        }

        let tol_d = get_dimension_tolerance(n_u64);

        let (v_norm, factor) = {
            let x_slice = unsafe { std::slice::from_raw_parts(x_view.ptr, n) };
            let y_slice = unsafe { std::slice::from_raw_parts(y_view.ptr, n) };
            let v_slice = unsafe { std::slice::from_raw_parts(v_view.ptr, n) };

            let x_n_sq = dot_neumaier(x_slice, x_slice);
            let y_n_sq = dot_neumaier(y_slice, y_slice);
            if !x_n_sq.is_finite() || !y_n_sq.is_finite() {
                return POLYDIM_NONFINITE_INPUT;
            }
            if (x_n_sq - 1.0).abs() > tol_d || (y_n_sq - 1.0).abs() > tol_d {
                return POLYDIM_NOT_UNIT;
            }

            let v_n_sq = dot_neumaier(v_slice, v_slice);
            if !v_n_sq.is_finite() {
                return POLYDIM_NONFINITE_INPUT;
            }
            let v_n = v_n_sq.max(0.0).sqrt();

            let xt_v_val = dot_neumaier(x_slice, v_slice);
            if !xt_v_val.is_finite() {
                return POLYDIM_NONFINITE_INPUT;
            }
            let tol_tangency = tol_d * v_n.max(1.0);
            if xt_v_val.abs() > tol_tangency {
                return POLYDIM_NOT_TANGENT;
            }

            let cos_th = dot_neumaier(x_slice, y_slice);
            if !cos_th.is_finite() {
                return POLYDIM_NONFINITE_INPUT;
            }
            let cos_clamped = cos_th.clamp(-1.0, 1.0);

            if 1.0 + cos_clamped <= CUT_LOCUS_NORM_SQ {
                return POLYDIM_PATH_REQUIRED;
            }

            let yt_v_val = dot_neumaier(y_slice, v_slice);
            if !yt_v_val.is_finite() {
                return POLYDIM_NONFINITE_INPUT;
            }
            let fact = yt_v_val / (1.0 + cos_clamped);

            (v_n, fact)
        };

        let is_v_inplace = out_const == v_view.ptr;
        {
            let x_slice = unsafe { std::slice::from_raw_parts(x_view.ptr, n) };
            let y_slice = unsafe { std::slice::from_raw_parts(y_view.ptr, n) };

            if is_v_inplace {
                // Stacked Borrows safe: v_slice was dropped before taking out_slice
                let out_slice = unsafe { std::slice::from_raw_parts_mut(out_view.ptr, n) };
                for i in 0..n {
                    out_slice[i] -= factor * (x_slice[i] + y_slice[i]);
                }
            } else {
                let v_slice = unsafe { std::slice::from_raw_parts(v_view.ptr, n) };
                let out_slice = unsafe { std::slice::from_raw_parts_mut(out_view.ptr, n) };
                for i in 0..n {
                    out_slice[i] = v_slice[i] - factor * (x_slice[i] + y_slice[i]);
                }
            }
        }

        let y_slice = unsafe { std::slice::from_raw_parts(y_view.ptr, n) };
        let out_slice_ro = unsafe { std::slice::from_raw_parts(out_view.ptr, n) };
        let yt_out = dot_neumaier(y_slice, out_slice_ro);
        if !yt_out.is_finite() || yt_out.abs() > (tol_d * 100.0) * v_norm.max(1.0) {
            return POLYDIM_PRECISION_LOSS;
        }

        let out_norm_sq = dot_neumaier(out_slice_ro, out_slice_ro);
        if !out_norm_sq.is_finite() {
            return POLYDIM_PRECISION_LOSS;
        }
        let out_norm = out_norm_sq.max(0.0).sqrt();
        if (out_norm - v_norm).abs() > (tol_d * 100.0) * v_norm.max(1.0) {
            return POLYDIM_PRECISION_LOSS;
        }

        POLYDIM_OK
    }));

    result.unwrap_or(POLYDIM_PANIC_TRAPPED)
}

    desc: *const PmtpTensorDescriptor,
) -> i32 {
    let result = catch_unwind(AssertUnwindSafe(|| {
        if desc.is_null() {
            return POLYDIM_INVALID_ARGUMENT;
        }
        if (desc as usize) % align_of::<u64>() != 0 {
            return POLYDIM_ALIGNMENT_ERROR;
        }
        let d = unsafe { &*desc };
        if d.magic != PMTP_MAGIC {
            return POLYDIM_PMTP_SLAB_ERROR;
        }
        if (d.version != 26155 && d.version != 26154 && d.version != 26153 &&
            d.version != 26152 && d.version != 26151 && d.version != 26150) || d.encoded_size as usize != PMTP_DESCRIPTOR_SIZE {
            return POLYDIM_RANGE_ERROR;
        }
        if d.dtype_code != 1 {
            return POLYDIM_INVALID_ARGUMENT;
        }
        if d.ndim < 1 || d.ndim > 4 {
            return POLYDIM_RANGE_ERROR;
        }
        if d.byte_length == 0 {
            return POLYDIM_RANGE_ERROR;
        }
        for &r in d.reserved.iter() {
            if r != 0 {
                return POLYDIM_INVALID_ARGUMENT;
            }
        }
        let mut expected_elements: u64 = 1;
        for i in 0..(d.ndim as usize) {
            if d.shape[i] == 0 {
                return POLYDIM_RANGE_ERROR;
            }
            match expected_elements.checked_mul(d.shape[i]) {
                Some(val) => expected_elements = val,
                None => return POLYDIM_RANGE_ERROR,
            }
        }
        for i in (d.ndim as usize)..4 {
            if d.shape[i] != 0 {
                return POLYDIM_RANGE_ERROR;
            }
        }
        let expected_bytes = match expected_elements.checked_mul(size_of::<f64>() as u64) {
            Some(val) => val,
            None => return POLYDIM_RANGE_ERROR,
        };
        if expected_bytes != d.byte_length {
            return POLYDIM_LENGTH_MISMATCH;
        }
        // Memory span overflow protection
        if d.byte_offset.checked_add(d.byte_length).is_none() {
            return POLYDIM_RANGE_ERROR;
        }
        if d.byte_offset % (align_of::<f64>() as u64) != 0 {
            return POLYDIM_ALIGNMENT_ERROR;
        }
        POLYDIM_OK
    }));

    result.unwrap_or(POLYDIM_PANIC_TRAPPED)
}

__global__ void k_lyapunov_elemwise_solve(
    const double* __restrict__ eigvals,
    const double* __restrict__ Q_rhs_Q,

) {

__global__ void k_symmetrize(double* __restrict__ A, int k) {
    int r = blockIdx.y * blockDim.y + threadIdx.y;
    int c = blockIdx.x * blockDim.x + threadIdx.x;
    if (r < k && c < r) {
        double val = 0.5 * (A[r * k + c] + A[c * k + r]);
        A[r * k + c] = val;
        A[c * k + r] = val;
    }
}

__global__ void k_skew_part(double* __restrict__ A, int k) {
    int r = blockIdx.y * blockDim.y + threadIdx.y;
    int c = blockIdx.x * blockDim.x + threadIdx.x;
    if (r < k && c <= r) {
        if (r == c) {
            A[r * k + c] = 0.0;
        } else {
            double val = 0.5 * (A[r * k + c] - A[c * k + r]);
            A[r * k + c] = val;
            A[c * k + r] = -val;
        }
    }
}

__global__ void k_compute_ginv(
    const double* __restrict__ Q,
    const double* __restrict__ eigvals,

            sum += Q[r * k + i] * inv_lam * Q[c * k + i];

    }

__global__ void k_build_woodbury_system(
    const double* __restrict__ G,
    const double* __restrict__ XtP,
    const double* __restrict__ PtP,
    double* __restrict__ M,
    double* __restrict__ RHS,
    double step,
    double scale,
    int k
) {
    int r = blockIdx.y * blockDim.y + threadIdx.y;
    int c = blockIdx.x * blockDim.x + threadIdx.x;
    int k2 = 2 * k;

    if (r < k2 && c < k2) {
        double tau = step / scale;
        double half_tau = 0.5 * tau;
        double inv_scale = 1.0 / scale;
        double val = 0.0;

        if (r < k && c < k) {
            val = (r == c ? inv_scale : 0.0) - half_tau * XtP[r * k + c];
        } else if (r < k && c >= k) {
            int c_sub = c - k;
            val = half_tau * G[r * k + c_sub];
        } else if (r >= k && c < k) {
            int r_sub = r - k;
            val = -half_tau * PtP[r_sub * k + c];
        } else {
            int r_sub = r - k;
            int c_sub = c - k;
            val = (r_sub == c_sub ? inv_scale : 0.0) + half_tau * XtP[c_sub * k + r_sub];
        }
        M[r * k2 + c] = val;
    }

    if (r < k2 && c < k) {
        double tau = step / scale;
        if (r < k) {
            RHS[r * k + c] = tau * G[r * k + c];
        } else {
            int r_sub = r - k;
            RHS[r * k + c] = tau * XtP[c * k + r_sub];
        }
    }
}

__global__ void k_assemble_coefficients(
    const double* __restrict__ Ex,
    const double* __restrict__ Ez,
    const double* __restrict__ W,
    double* __restrict__ Ax,
    double* __restrict__ Az,
    int k
) {
    int r = blockIdx.y * blockDim.y + threadIdx.y;
    int c = blockIdx.x * blockDim.x + threadIdx.x;
    int k2 = 2 * k;

    if (r < k && c < k) {
        double ex_w1 = 0.0;
        double ez_w1 = 0.0;
        for (int i = 0; i < k; ++i) {
            double w1_val = W[i * k + c];
            ex_w1 += Ex[r * k + i] * w1_val;
            ez_w1 += Ez[r * k + i] * w1_val;
        }
        double w2_val = W[(k + r) * k + c];
        Ax[r * k + c] = (r == c ? 1.0 : 0.0) + ex_w1 - w2_val;
        Az[r * k + c] = ez_w1;
    }
}

    }
    if (!std::isfinite(step)) {
        return polydim_v26150::POLYDIM_INVALID_ARGUMENT;
    }

    // In production, orchestration is handled by cuBLAS/cuSOLVER on stream
    return polydim_v26150::POLYDIM_OK;
}

    @triton.jit
    def _sphere_project_triton_kernel(
        x_ptr,
        out_ptr,
        n_elements,
        norm_inv: tl.float64,
        BLOCK_SIZE: tl.constexpr
    ):
        pid = tl.program_id(axis=0)
        block_start = pid * BLOCK_SIZE
        offsets = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offsets < n_elements

        x_ptr,
        y_ptr,
        v_ptr,
        out_ptr,
        factor: tl.float64,
        n_elements,
        BLOCK_SIZE: tl.constexpr
    ):
        pid = tl.program_id(axis=0)

        offsets = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offsets < n_elements

        y = tl.load(y_ptr + offsets, mask=mask, other=0.0)
        v = tl.load(v_ptr + offsets, mask=mask, other=0.0)

        # Levi-Civita parallel transport update: out = v - factor * (x + y)
        out = v - factor * (x + y)
        tl.store(out_ptr + offsets, out, mask=mask)


def sphere_project_triton(x: Any, out: Optional[Any] = None) -> Tuple[int, Any, Dict[str, Any]]:
    if not HAS_TRITON:
        return POLYDIM_FP_ENV_ERROR, None, {"error": "Triton not available on host runtime"}

    if not isinstance(x, torch.Tensor):
        return POLYDIM_INVALID_ARGUMENT, None, {"error": "Expected torch.Tensor"}

    if torch.is_complex(x):
        return POLYDIM_INVALID_ARGUMENT, None, {"error": "Complex dtypes unsupported"}

    if x.dtype != torch.float64:
        return POLYDIM_INVALID_ARGUMENT, None, {"error": "Expected torch.float64 dtype"}

    if not x.is_cuda and x.device.type not in ("cuda", "xpu"):
        return POLYDIM_FP_ENV_ERROR, None, {"error": f"Tensor must be on accelerator device, got {x.device}"}

    if not x.is_contiguous():
        x = x.contiguous()

    n = x.numel()
    if n == 0 or n > 10_000_000:
        return POLYDIM_RANGE_ERROR, None, {"error": f"Invalid dimension n={n}"}

    if out is None:
        out = torch.empty_like(x)
    else:
        if not isinstance(out, torch.Tensor) or out.shape != x.shape or out.dtype != x.dtype:
            return POLYDIM_INVALID_ARGUMENT, None, {"error": "Invalid out tensor shape/dtype"}
        if out.device != x.device:
            return POLYDIM_INVALID_ARGUMENT, None, {"error": f"Device mismatch: out ({out.device}) != x ({x.device})"}
        if not out.is_contiguous():
            return POLYDIM_INVALID_ARGUMENT, None, {"error": "out tensor must be contiguous"}
        if out.data_ptr() != x.data_ptr():
            start_x = x.data_ptr()
            end_x = start_x + x.numel() * 8
            start_out = out.data_ptr()
            end_out = start_out + out.numel() * 8
            if max(start_x, start_out) < min(end_x, end_out):
                return POLYDIM_ALIAS_VIOLATION, None, {"error": "out tensor partially aliases x"}

    # Two-pass scaled norm without overflowing max_val * scaled_norm
    with torch.no_grad():
        abs_x = torch.abs(x)
        max_val = float(torch.max(abs_x).item())

        if not math.isfinite(max_val):
            return POLYDIM_NONFINITE_INPUT, None, {"error": "Non-finite input elements"}
        if max_val == 0.0:
            return POLYDIM_ZERO_VECTOR, None, {"error": "Zero vector input"}

        scaled_x_pre = x / max_val
        scaled_norm = float(torch.norm(scaled_x_pre).item())

        if not math.isfinite(scaled_norm) or scaled_norm == 0.0:
            return POLYDIM_NUMERICAL_FAILURE, None, {"error": "Invalid scaled norm"}

        norm_val = max_val * scaled_norm
        inv_norm = float(1.0 / norm_val)
        if not math.isfinite(inv_norm) or inv_norm <= 0.0:
            return POLYDIM_NUMERICAL_FAILURE, None, {"error": "Invalid combined norm calculation"}

        BLOCK_SZ = 1024
        grid = (triton.cdiv(n, BLOCK_SZ),)

        try:
            _sphere_project_triton_kernel[grid](
                x, out, n, inv_norm, BLOCK_SIZE=BLOCK_SZ
            )
        except Exception as e:
            return POLYDIM_NUMERICAL_FAILURE, None, {"error": f"Triton kernel launch failed: {e}"}

    return POLYDIM_OK, out, {"max_val": max_val, "scaled_norm": scaled_norm, "backend": "triton"}


def sphere_transport_triton(
    x: Any,
    y: Any,
    v: Any,
    out: Optional[Any] = None
) -> Tuple[int, Any, Dict[str, Any]]:
    if not HAS_TRITON:
        return POLYDIM_FP_ENV_ERROR, None, {"error": "Triton not available on host runtime"}

    for t, name in [(x, "x"), (y, "y"), (v, "v")]:
        if not isinstance(t, torch.Tensor) or t.dtype != torch.float64:
            return POLYDIM_INVALID_ARGUMENT, None, {"error": f"Expected torch.float64 for {name}"}
        if not t.is_cuda and t.device.type not in ("cuda", "xpu"):
            return POLYDIM_FP_ENV_ERROR, None, {"error": f"{name} must be on accelerator device"}

    n = x.numel()
    if y.numel() != n or v.numel() != n:
        return POLYDIM_LENGTH_MISMATCH, None, {"error": "Dimension mismatch between tensors"}
    if n == 0 or n > 10_000_000:
        return POLYDIM_RANGE_ERROR, None, {"error": f"Invalid dimension n={n}"}

    if out is None:
        out = torch.empty_like(v)

    with torch.no_grad():
        cos_th = float(torch.dot(x, y).item())
        if not math.isfinite(cos_th):
            return POLYDIM_NONFINITE_INPUT, None, {"error": "Non-finite inner product x^T y"}

        if 1.0 + cos_th <= 1.0e-8:
            return POLYDIM_PATH_REQUIRED, None, {"error": "Antipodal cut locus path required"}

        yt_v = float(torch.dot(y, v).item())
        if not math.isfinite(yt_v):
            return POLYDIM_NONFINITE_INPUT, None, {"error": "Non-finite inner product y^T v"}

        factor = yt_v / (1.0 + cos_th)

        BLOCK_SZ = 1024
        grid = (triton.cdiv(n, BLOCK_SZ),)

        try:
            _sphere_transport_triton_kernel[grid](
                x, y, v, out, factor, n, BLOCK_SIZE=BLOCK_SZ
            )
        except Exception as e:
            return POLYDIM_NUMERICAL_FAILURE, None, {"error": f"Triton transport launch failed: {e}"}

    return POLYDIM_OK, out, {"factor": factor, "backend": "triton"}
