// Neumaier Compensated Summation
#[inline(always)]
pub fn neumaier_sum(slice: &[f64]) -> f64 {
    let mut sum = 0.0f64;
    let mut comp = 0.0f64;
    for &val in slice {
        let t = sum + val;
        if sum.abs() >= val.abs() {
            comp += (sum - t) + val;
        } else {
            comp += (val - t) + sum;
        }
        sum = t;
    }
    sum + comp
}

#[inline(always)]
pub fn dot_neumaier(a: &[f64], b: &[f64]) -> f64 {
    let mut sum = 0.0f64;
    let mut comp = 0.0f64;
    for (&x, &y) in a.iter().zip(b.iter()) {
        let prod = x * y;
        let t = sum + prod;
        if sum.abs() >= prod.abs() {
            comp += (sum - t) + prod;
        } else {
            comp += (prod - t) + sum;
        }
        sum = t;
    }
    sum + comp
}

// Spherical Projection: y = x / ||x||_2
#[no_mangle]
pub unsafe extern "C" fn polydim_sphere_project_v1350(
    in_view: PolydimF64ConstView,
    out_view: PolydimF64MutView,
) -> i32 {
    let result = catch_unwind(AssertUnwindSafe(|| {
        if in_view.ptr.is_null() || out_view.ptr.is_null() {
            return POLYDIM_INVALID_ARGUMENT;
        }
        if in_view.len != out_view.len {
            return POLYDIM_LENGTH_MISMATCH;
        }
        let n = in_view.len;
        if n == 0 || n > MAX_DIMENSION || n > (isize::MAX as u64) {
            return POLYDIM_RANGE_ERROR;
        }

        let in_slice = unsafe { std::slice::from_raw_parts(in_view.ptr, n as usize) };
        let out_slice = unsafe { std::slice::from_raw_parts_mut(out_view.ptr, n as usize) };

        let norm_sq = dot_neumaier(in_slice, in_slice);
        if !norm_sq.is_finite() {
            return POLYDIM_NONFINITE_INPUT;
        }
        if norm_sq == 0.0 {
            return POLYDIM_ZERO_VECTOR;
        }

        let norm = norm_sq.sqrt();
        let inv_norm = 1.0 / norm;

        for (dst, &src) in out_slice.iter_mut().zip(in_slice.iter()) {
            let val = src * inv_norm;
            if src != 0.0 && val.is_subnormal() {
                return POLYDIM_UNREPRESENTABLE_OUTPUT;
            }
            *dst = val;
        }

        POLYDIM_OK
    }));

    result.unwrap_or(POLYDIM_PANIC_TRAPPED)
}

// PMTP Descriptor Validation
#[no_mangle]
pub unsafe extern "C" fn polydim_pmtp_validate_descriptor_v1350(
    desc: *const PmtpTensorDescriptor,
) -> i32 {
    if desc.is_null() {
        return POLYDIM_INVALID_ARGUMENT;
    }
    let d = unsafe { &*desc };
    if d.magic != PMTP_MAGIC {
        return POLYDIM_PMTP_SLAB_ERROR;
    }
    if d.version != 1350 || d.encoded_size as usize != PMTP_DESCRIPTOR_SIZE {
        return POLYDIM_RANGE_ERROR;
    }
    if d.ndim > 4 {
        return POLYDIM_RANGE_ERROR;
    }
    if d.byte_length == 0 {
        return POLYDIM_ZERO_VECTOR;
    }
    POLYDIM_OK
}
