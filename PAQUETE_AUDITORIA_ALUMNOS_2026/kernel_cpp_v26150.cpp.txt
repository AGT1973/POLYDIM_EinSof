// Unified Status Codes (0 to -15, -99)
enum Status : std::int32_t {
    POLYDIM_OK                     = 0,
    POLYDIM_INVALID_ARGUMENT       = -1,
    POLYDIM_NONFINITE_INPUT        = -2,
    POLYDIM_ALIAS_VIOLATION        = -3,
    POLYDIM_ALIGNMENT_ERROR        = -4,
    POLYDIM_NOT_UNIT               = -5,
    POLYDIM_NOT_TANGENT            = -6,
    POLYDIM_ZERO_VECTOR            = -7,
    POLYDIM_PATH_REQUIRED          = -8,
    POLYDIM_NUMERICAL_FAILURE      = -9,
    POLYDIM_RANGE_ERROR            = -10,

    POLYDIM_LENGTH_MISMATCH        = -13,

    POLYDIM_PMTP_SLAB_ERROR        = -15,

constexpr std::uint64_t kMaxDimension = 10'000'000;
constexpr double kUnitTolerance       = 1.0e-12;
constexpr double kTangencyTolerance   = 1.0e-12;
constexpr double kOutputTolerance     = 1.0e-10;
constexpr double kCutLocusNormSq      = 1.0e-8;

#pragma pack(push, 1)
struct PmtpTensorDescriptor {
    char          magic[4];       // "PDM4"
    std::uint16_t version;        // 1350
    std::uint16_t encoded_size;   // 88
    std::uint64_t segment_id;
    std::uint64_t buffer_id;
    std::uint64_t generation;
    std::uint64_t byte_offset;
    std::uint64_t byte_length;
    std::uint8_t  dtype_code;     // 1 = f64
    std::uint8_t  ndim;
    std::uint8_t  reserved[6];
    std::uint64_t shape[4];
};
#pragma pack(pop)

static_assert(sizeof(PmtpTensorDescriptor) == 88, "PmtpTensorDescriptor must be exactly 88 bytes");

extern "C" {
    struct PolydimF64ConstView {
        const double* ptr;
        std::uint64_t len;
    };

    struct PolydimF64MutView {
        double* ptr;
        std::uint64_t len;
    };
}

// Neumaier compensated summation for exact inner products
inline void neumaier_accumulate(double val, double& sum, double& comp) noexcept {
    double t = sum + val;
    if (std::abs(sum) >= std::abs(val)) {
        comp += (sum - t) + val;
    } else {
        comp += (val - t) + sum;
    }
    sum = t;
}

inline double dot_neumaier(const double* a, const double* b, std::uint64_t n) noexcept {
    double sum = 0.0;
    double comp = 0.0;
    for (std::uint64_t i = 0; i < n; ++i) {
        double prod = a[i] * b[i];
        neumaier_accumulate(prod, sum, comp);
    }
    return sum + comp;
}

// Exact Antipodal Axis Selection: j = argmin |x_i|
inline std::uint64_t find_min_abs_axis(const double* x, std::uint64_t n) noexcept {
    std::uint64_t min_idx = 0;
    double min_val = std::abs(x[0]);
    for (std::uint64_t i = 1; i < n; ++i) {
        double val = std::abs(x[i]);
        if (val < min_val) {
            min_val = val;
            min_idx = i;
        }
    }
    return min_idx;
}

// Spherical Projection: y = x / ||x||_2
POLYDIM_EXPORT std::int32_t polydim_sphere_project_v1350(
    PolydimF64ConstView in_view,
    PolydimF64MutView out_view
) noexcept {
    if (!in_view.ptr || !out_view.ptr) return POLYDIM_INVALID_ARGUMENT;
    if (in_view.len != out_view.len)   return POLYDIM_LENGTH_MISMATCH;
    if (in_view.len == 0 || in_view.len > kMaxDimension) return POLYDIM_RANGE_ERROR;

    // Check pointer alias violation
    if (in_view.ptr != out_view.ptr) {
        const char* in_b = reinterpret_cast<const char*>(in_view.ptr);
        const char* in_e = in_b + in_view.len * sizeof(double);
        const char* out_b = reinterpret_cast<const char*>(out_view.ptr);
        const char* out_e = out_b + out_view.len * sizeof(double);
        if (std::max(in_b, out_b) < std::min(in_e, out_e)) {
            return POLYDIM_ALIAS_VIOLATION;
        }
    }

    std::uint64_t n = in_view.len;
    double norm_sq = dot_neumaier(in_view.ptr, in_view.ptr, n);

    if (!std::isfinite(norm_sq)) return POLYDIM_NONFINITE_INPUT;
    if (norm_sq == 0.0)         return POLYDIM_ZERO_VECTOR;

    double norm = std::sqrt(norm_sq);
    double inv_norm = 1.0 / norm;

    // Transactional write
    for (std::uint64_t i = 0; i < n; ++i) {
        double val = in_view.ptr[i] * inv_norm;
        if (in_view.ptr[i] != 0.0 && std::fpclassify(val) == FP_SUBNORMAL) {
            return POLYDIM_UNREPRESENTABLE_OUTPUT;
        }
        out_view.ptr[i] = val;
    }

    return POLYDIM_OK;
}

// Spherical Parallel Transport on S^(D-1)
POLYDIM_EXPORT std::int32_t polydim_sphere_transport_v1350(
    PolydimF64ConstView x_view,
    PolydimF64ConstView y_view,
    PolydimF64ConstView v_view,
    PolydimF64MutView out_view
) noexcept {
    if (!x_view.ptr || !y_view.ptr || !v_view.ptr || !out_view.ptr) return POLYDIM_INVALID_ARGUMENT;
    std::uint64_t n = x_view.len;
    if (y_view.len != n || v_view.len != n || out_view.len != n) return POLYDIM_LENGTH_MISMATCH;
    if (n == 0 || n > kMaxDimension) return POLYDIM_RANGE_ERROR;

    // Check unit norm of base points
    double x_norm_sq = dot_neumaier(x_view.ptr, x_view.ptr, n);
    double y_norm_sq = dot_neumaier(y_view.ptr, y_view.ptr, n);
    if (std::abs(x_norm_sq - 1.0) > kUnitTolerance || std::abs(y_norm_sq - 1.0) > kUnitTolerance) {
        return POLYDIM_NOT_UNIT;
    }

    // Check tangency: x^T v == 0
    double xt_v = dot_neumaier(x_view.ptr, v_view.ptr, n);
    if (std::abs(xt_v) > kTangencyTolerance) {
        return POLYDIM_NOT_TANGENT;
    }

    // Dot product: cos(theta) = x^T y
    double cos_th = dot_neumaier(x_view.ptr, y_view.ptr, n);
    cos_th = std::clamp(cos_th, -1.0, 1.0);

    // Antipodal check
    if (1.0 + cos_th <= kCutLocusNormSq) {
        return POLYDIM_PATH_REQUIRED;
    }

    // Standard Levi-Civita Transport:
    // P_x->y(v) = v - (y^T v)/(1 + cos(th)) * (x + y)
    double yt_v = dot_neumaier(y_view.ptr, v_view.ptr, n);
    double factor = yt_v / (1.0 + cos_th);

    for (std::uint64_t i = 0; i < n; ++i) {
        double transported = v_view.ptr[i] - factor * (x_view.ptr[i] + y_view.ptr[i]);
        out_view.ptr[i] = transported;
    }

    // Output tangency audit: y^T out == 0
    double yt_out = dot_neumaier(y_view.ptr, out_view.ptr, n);
    if (std::abs(yt_out) > kOutputTolerance) {
        return POLYDIM_PRECISION_LOSS;
    }

    return POLYDIM_OK;
}

// PMTP Descriptor Validation
POLYDIM_EXPORT std::int32_t polydim_pmtp_validate_descriptor_v1350(
    const PmtpTensorDescriptor* desc
) noexcept {
    if (!desc) return POLYDIM_INVALID_ARGUMENT;
    if (std::memcmp(desc->magic, "PDM4", 4) != 0) return POLYDIM_PMTP_SLAB_ERROR;
    if (desc->version != 1350) return POLYDIM_RANGE_ERROR;
    if (desc->encoded_size != sizeof(PmtpTensorDescriptor)) return POLYDIM_LENGTH_MISMATCH;
    if (desc->ndim > 4) return POLYDIM_RANGE_ERROR;
    if (desc->byte_length == 0) return POLYDIM_ZERO_VECTOR;
    return POLYDIM_OK;
}
